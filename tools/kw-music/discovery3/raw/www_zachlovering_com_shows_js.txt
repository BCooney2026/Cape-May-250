const MON = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
function utcDate(y, m, d) { return new Date(Date.UTC(y, m - 1, d)); }
function addDays(d, n) { return new Date(d.getTime() + n * 864e5); }
function todayKeyWest() {
  const p = new Intl.DateTimeFormat("en-US", { timeZone: "America/New_York", year: "numeric", month: "numeric", day: "numeric" }).formatToParts(new Date());
  const g = t => +p.find(x => x.type === t).value;
  return new Date(Date.UTC(g("year"), g("month") - 1, g("day")));
}
function buildWeeks() {
  return weeksData.filter(w => w.shows && w.shows.length).map(w => {
    const start = utcDate(...w.start);
    return {
      start,
      end: addDays(start, 6),
      days: w.shows.map(s => ({ ...s, date: addDays(start, DAYS.indexOf(s.day)) }))
    };
  });
}
function shortRange(a, b) {
  if (a.getUTCMonth() === b.getUTCMonth()) return `${MON[a.getUTCMonth()]} ${a.getUTCDate()}–${b.getUTCDate()}`;
  return `${MON[a.getUTCMonth()]} ${a.getUTCDate()} – ${MON[b.getUTCMonth()]} ${b.getUTCDate()}`;
}
function rangeLabel(a, b) {
  const y = b.getUTCFullYear();
  if (a.getUTCMonth() === b.getUTCMonth()) return `${MON[a.getUTCMonth()]} ${a.getUTCDate()} – ${b.getUTCDate()}, ${y}`;
  return `${MON[a.getUTCMonth()]} ${a.getUTCDate()}${a.getUTCFullYear() !== y ? ", " + a.getUTCFullYear() : ""} – ${MON[b.getUTCMonth()]} ${b.getUTCDate()}, ${y}`;
}
function isThisWeek(wk, today) {
  return today >= wk.start && today <= wk.end;
}
if (typeof document !== "undefined") {
  const weeks = buildWeeks();
  const today = todayKeyWest();
  let cur = Math.max(0, weeks.findIndex(wk => today <= wk.end));
  const $ = id => document.getElementById(id);
  function render() {
    const wk = weeks[cur];
    const label = (isThisWeek(wk, today) || (today < weeks[0].start && cur === 0) ? "This week in Key West, FL · " : "Key West, FL · ") + rangeLabel(wk.start, wk.end);
    $("week-label").textContent = label;
    $("show-list").innerHTML = wk.days.map(s =>
      `<li class="${s.off ? "off" : ""}"><div class="d">${s.day}<span>${MON[s.date.getUTCMonth()]} ${s.date.getUTCDate()}${s.time ? " · " + s.time : ""}</span></div><div class="v"><b>${s.venue}</b><span>${s.note ? s.note + " · " : ""}${s.place || "Key West, FL"}</span></div></li>`).join("");
    $("week-prev").disabled = cur === 0;
    $("week-next").disabled = cur === weeks.length - 1;
    const strip = $("week-strip");
    if (strip) {
      strip.innerHTML = weeks.map((w, i) =>
        `<button type="button" class="week-tab${i === cur ? " on" : ""}" data-week="${i}" aria-pressed="${i === cur}" aria-label="${rangeLabel(w.start, w.end)}">${shortRange(w.start, w.end)}</button>`
      ).join("");
    }
  }
  $("week-prev").addEventListener("click", () => { if (cur > 0) { cur--; render(); } });
  $("week-next").addEventListener("click", () => { if (cur < weeks.length - 1) { cur++; render(); } });
  const stripEl = $("week-strip");
  if (stripEl) stripEl.addEventListener("click", e => {
    const btn = e.target.closest("[data-week]");
    if (!btn) return;
    cur = +btn.getAttribute("data-week");
    render();
  });
  render();
  $("yr").textContent = new Date().getFullYear();
} else if (typeof module !== "undefined") module.exports = { buildWeeks, rangeLabel, shortRange };
